# Serving dependencies for the FastAPI backend.
#
# Deliberately tiny. The API reads a prebuilt SQLite snapshot in which every
# prediction has already been computed, so it loads no model and touches no
# DataFrame. Verified: importing app.main pulls in zero of pandas, numpy,
# scikit-learn, scipy, xgboost, pyarrow or nba_api.
#
# Measured: this set installs to ~21 MB (20 packages) versus ~400 MB with the
# training stack, which matters on Render's 512 MB free tier. Anything needed to
# *build* a snapshot or train a model lives in requirements-data.txt instead.
#
# sqlite3 is in the standard library, so there is no database driver here.

fastapi==0.128.8
uvicorn==0.39.0
pydantic==2.12.5

# Only used to download the snapshot at startup when SNAPSHOT_URL is set.
requests==2.32.5
